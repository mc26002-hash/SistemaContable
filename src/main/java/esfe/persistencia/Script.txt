USE ContabilidadESFE;
GO


/* =====================================================
   NIVEL 1: TABLAS INDEPENDIENTES
   ===================================================== */

CREATE TABLE Roles (
    RolId INT IDENTITY(1,1) PRIMARY KEY,
    NombreRol NVARCHAR(50) NOT NULL UNIQUE,
    Descripcion NVARCHAR(250) NULL,
    Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE TiposCuenta (
    TipoCuentaId INT IDENTITY(1,1) PRIMARY KEY,
    NombreTipo NVARCHAR(50) NOT NULL UNIQUE,
    Naturaleza CHAR(1) NOT NULL,
    CONSTRAINT CK_TiposCuenta_Naturaleza
        CHECK (Naturaleza IN ('D','H'))
);
GO

CREATE TABLE TiposPartida (
    TipoPartidaId INT IDENTITY(1,1) PRIMARY KEY,
    NombreTipo NVARCHAR(50) NOT NULL UNIQUE
);
GO

CREATE TABLE TiposDocumentoFiscal (
    TipoDocumentoFiscalId INT IDENTITY(1,1) PRIMARY KEY,
    Codigo NVARCHAR(10) NOT NULL UNIQUE,
    Nombre NVARCHAR(100) NOT NULL
);
GO


/* =====================================================
   NIVEL 2: TABLAS QUE DEPENDEN DE NIVEL 1
   ===================================================== */

CREATE TABLE Usuarios (
    UsuarioId INT IDENTITY(1,1) PRIMARY KEY,
    RolId INT NOT NULL,
    NombreUsuario NVARCHAR(50) NOT NULL,
    PasswordHash VARBINARY(256) NOT NULL,
    NombreCompleto NVARCHAR(150) NOT NULL,
    CorreoElectronico NVARCHAR(150) NOT NULL,
    Activo BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_Usuarios_Roles
        FOREIGN KEY (RolId) REFERENCES Roles(RolId),

    CONSTRAINT UQ_Usuarios_NombreUsuario
        UNIQUE (NombreUsuario),

    CONSTRAINT UQ_Usuarios_Correo
        UNIQUE (CorreoElectronico)
);
GO

CREATE TABLE CentrosCosto (
    CentroCostoId INT IDENTITY(1,1) PRIMARY KEY,
    Codigo NVARCHAR(20) NOT NULL UNIQUE,
    Nombre NVARCHAR(100) NOT NULL,
    Responsable NVARCHAR(150) NULL,
    Presupuesto DECIMAL(18,2) NOT NULL DEFAULT 0,
    Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE CuentasContables (
    CuentaId INT IDENTITY(1,1) PRIMARY KEY,
    TipoCuentaId INT NOT NULL,
    CuentaPadreId INT NULL,
    CodigoCuenta NVARCHAR(20) NOT NULL UNIQUE,
    NombreCuenta NVARCHAR(150) NOT NULL,
    PermiteMovimiento BIT NOT NULL DEFAULT 1,
    Activa BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_CuentasContables_TiposCuenta
        FOREIGN KEY (TipoCuentaId) REFERENCES TiposCuenta(TipoCuentaId),

    CONSTRAINT FK_CuentasContables_CuentaPadre
        FOREIGN KEY (CuentaPadreId) REFERENCES CuentasContables(CuentaId)
);
GO

CREATE TABLE Terceros (
    TerceroId INT IDENTITY(1,1) PRIMARY KEY,
    TipoTercero NVARCHAR(20) NOT NULL,
    Nombre NVARCHAR(150) NOT NULL,
    NIT NVARCHAR(20) NULL,
    NRC NVARCHAR(20) NULL,
    Correo NVARCHAR(150) NULL,
    Telefono NVARCHAR(30) NULL,
    Activo BIT NOT NULL DEFAULT 1,

    CONSTRAINT CK_Terceros_TipoTercero
        CHECK (TipoTercero IN ('Cliente','Proveedor','Ambos'))
);
GO


/* =====================================================
   NIVEL 3: PERÍODOS CONTABLES
   ===================================================== */

CREATE TABLE PeriodosContables (
    PeriodoId INT IDENTITY(1,1) PRIMARY KEY,
    Anio INT NOT NULL,
    Mes INT NOT NULL,
    Estado NVARCHAR(20) NOT NULL DEFAULT 'Abierto',
    CerradoPorUsuarioId INT NULL,
    FechaCierre DATETIME2 NULL,

    CONSTRAINT FK_PeriodosContables_Usuarios
        FOREIGN KEY (CerradoPorUsuarioId) REFERENCES Usuarios(UsuarioId),

    CONSTRAINT CK_PeriodosContables_Mes
        CHECK (Mes BETWEEN 1 AND 12),

    CONSTRAINT CK_PeriodosContables_Estado
        CHECK (Estado IN ('Abierto','Cerrado')),

    CONSTRAINT UQ_PeriodosContables_Anio_Mes
        UNIQUE (Anio, Mes)
);
GO


/* =====================================================
   NIVEL 4: PARTIDAS CONTABLES
   ===================================================== */

CREATE TABLE PartidasContables (
    PartidaId INT IDENTITY(1,1) PRIMARY KEY,
    PeriodoId INT NOT NULL,
    TipoPartidaId INT NOT NULL,
    NumeroPartida NVARCHAR(30) NOT NULL UNIQUE,
    Fecha DATE NOT NULL,
    ConceptoGeneral NVARCHAR(250) NOT NULL,
    Estado NVARCHAR(20) NOT NULL DEFAULT 'Borrador',
    CreadaPorUsuarioId INT NOT NULL,
    FechaCreacion DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    AnuladaPorUsuarioId INT NULL,
    FechaAnulacion DATETIME2 NULL,
    MotivoAnulacion NVARCHAR(300) NULL,

    CONSTRAINT FK_PartidasContables_Periodos
        FOREIGN KEY (PeriodoId) REFERENCES PeriodosContables(PeriodoId),

    CONSTRAINT FK_PartidasContables_TiposPartida
        FOREIGN KEY (TipoPartidaId) REFERENCES TiposPartida(TipoPartidaId),

    CONSTRAINT FK_PartidasContables_UsuarioCreacion
        FOREIGN KEY (CreadaPorUsuarioId) REFERENCES Usuarios(UsuarioId),

    CONSTRAINT FK_PartidasContables_UsuarioAnulacion
        FOREIGN KEY (AnuladaPorUsuarioId) REFERENCES Usuarios(UsuarioId),

    CONSTRAINT CK_PartidasContables_Estado
        CHECK (Estado IN ('Borrador','Contabilizada','Anulada'))
);
GO

CREATE TABLE DetallesPartida (
    DetallePartidaId INT IDENTITY(1,1) PRIMARY KEY,
    PartidaId INT NOT NULL,
    CuentaId INT NOT NULL,
    CentroCostoId INT NULL,
    Descripcion NVARCHAR(250) NULL,
    Debe DECIMAL(18,2) NOT NULL DEFAULT 0,
    Haber DECIMAL(18,2) NOT NULL DEFAULT 0,

    CONSTRAINT FK_DetallesPartida_Partidas
        FOREIGN KEY (PartidaId) REFERENCES PartidasContables(PartidaId),

    CONSTRAINT FK_DetallesPartida_Cuentas
        FOREIGN KEY (CuentaId) REFERENCES CuentasContables(CuentaId),

    CONSTRAINT FK_DetallesPartida_CentrosCosto
        FOREIGN KEY (CentroCostoId) REFERENCES CentrosCosto(CentroCostoId),

    CONSTRAINT CK_DetallesPartida_DebeHaber
        CHECK (
            Debe >= 0 AND Haber >= 0
            AND (
                (Debe > 0 AND Haber = 0)
                OR
                (Debe = 0 AND Haber > 0)
            )
        )
);
GO


/* =====================================================
   NIVEL 5: DOCUMENTOS FISCALES Y RETENCIONES
   ===================================================== */

CREATE TABLE DocumentosFiscales (
    DocumentoFiscalId INT IDENTITY(1,1) PRIMARY KEY,
    TerceroId INT NOT NULL,
    TipoDocumentoFiscalId INT NOT NULL,
    PartidaId INT NULL,
    TipoLibro NVARCHAR(20) NOT NULL,
    NumeroDocumento NVARCHAR(50) NOT NULL,
    FechaDocumento DATE NOT NULL,
    MontoExento DECIMAL(18,2) NOT NULL DEFAULT 0,
    MontoGravado DECIMAL(18,2) NOT NULL DEFAULT 0,
    IVA DECIMAL(18,2) NOT NULL DEFAULT 0,
    Total DECIMAL(18,2) NOT NULL,
    AplicaRetencion BIT NOT NULL DEFAULT 0,

    CONSTRAINT FK_DocumentosFiscales_Terceros
        FOREIGN KEY (TerceroId) REFERENCES Terceros(TerceroId),

    CONSTRAINT FK_DocumentosFiscales_TiposDocumento
        FOREIGN KEY (TipoDocumentoFiscalId) REFERENCES TiposDocumentoFiscal(TipoDocumentoFiscalId),

    CONSTRAINT FK_DocumentosFiscales_Partidas
        FOREIGN KEY (PartidaId) REFERENCES PartidasContables(PartidaId),

    CONSTRAINT CK_DocumentosFiscales_TipoLibro
        CHECK (TipoLibro IN ('Compra','Venta')),

    CONSTRAINT CK_DocumentosFiscales_Montos
        CHECK (
            MontoExento >= 0
            AND MontoGravado >= 0
            AND IVA >= 0
            AND Total >= 0
        )
);
GO

CREATE TABLE Retenciones (
    RetencionId INT IDENTITY(1,1) PRIMARY KEY,
    DocumentoFiscalId INT NOT NULL,
    TipoRetencion NVARCHAR(20) NOT NULL,
    Porcentaje DECIMAL(5,2) NOT NULL,
    MontoBase DECIMAL(18,2) NOT NULL,
    MontoRetenido DECIMAL(18,2) NOT NULL,

    CONSTRAINT FK_Retenciones_DocumentosFiscales
        FOREIGN KEY (DocumentoFiscalId) REFERENCES DocumentosFiscales(DocumentoFiscalId),

    CONSTRAINT CK_Retenciones_TipoRetencion
        CHECK (TipoRetencion IN ('IVA','ISR')),

    CONSTRAINT CK_Retenciones_Montos
        CHECK (
            Porcentaje > 0
            AND MontoBase >= 0
            AND MontoRetenido >= 0
        )
);
GO


/* =====================================================
   NIVEL 6: AUDITORÍA SIMPLE
   ===================================================== */

CREATE TABLE ReaperturasPeriodo (
    ReaperturaPeriodoId INT IDENTITY(1,1) PRIMARY KEY,
    PeriodoId INT NOT NULL,
    UsuarioId INT NOT NULL,
    Motivo NVARCHAR(300) NOT NULL,
    FechaReapertura DATETIME2 NOT NULL DEFAULT SYSDATETIME(),

    CONSTRAINT FK_ReaperturasPeriodo_Periodos
        FOREIGN KEY (PeriodoId) REFERENCES PeriodosContables(PeriodoId),

    CONSTRAINT FK_ReaperturasPeriodo_Usuarios
        FOREIGN KEY (UsuarioId) REFERENCES Usuarios(UsuarioId)
);
GO

CREATE TABLE BitacoraSistema (
    BitacoraId BIGINT IDENTITY(1,1) PRIMARY KEY,
    UsuarioId INT NULL,
    Accion NVARCHAR(50) NOT NULL,
    TablaAfectada NVARCHAR(100) NOT NULL,
    Descripcion NVARCHAR(500) NOT NULL,
    FechaHora DATETIME2 NOT NULL DEFAULT SYSDATETIME(),

    CONSTRAINT FK_BitacoraSistema_Usuarios
        FOREIGN KEY (UsuarioId) REFERENCES Usuarios(UsuarioId)
);
GO
